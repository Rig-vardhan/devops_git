pipeline {
    agent any

    stages {
        stage('Checkout') {
            steps {
                checkout([$class: 'GitSCM', branches: [[name: 'main']], extensions: [], userRemoteConfigs: [[url: 'https://github.com/Rig-vardhan/devops_git']]])
            }
        }
        stage('Build') {
            steps {
                git branch: 'main', url: 'https://github.com/Rig-vardhan/devops_git'
                bat 'py intro.py'
            }
        }
        stage('Test') {
            steps {
                echo 'TESTING'
            }
        }
         stage('Docker Deploy') {
            steps {
                echo 'Building Docker image...'
                // Build Docker image (tag as 'myapp:latest')
                bat 'docker build -t intro:latest .'

                echo 'Running Docker container...'
                // Run Docker container on port 5000
                bat 'docker run -d --name intro_container intro:latest'

                echo 'Docker deployment completed!'
            }
        }
    }
}
